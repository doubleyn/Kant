```bash

vi pyproject.toml
# requires-python = ">=3.12" 로 수정 후

# Updated `.python-version` from `3.13` -> `3.12`

uv run main.py
# Using CPython 3.12.16
# Removed virtual environment at: .venv
# Creating virtual environment at: .venv
```
